### 大致思路
对于每个成员$i$，有三个部门可供选择。先假设我们全部选择第二大的，然后再考虑假如选择可以对于原结果额外做出的的更多贡献的。对于每个$i$，求最大和最小二$a_i$相对第二大$a_i$的差值，对于最大差值做贪心即可。
### 实现
具体来说，可以在输入时对于每个$i$，先把三个对应的$a_i$挨个减去第二大的$a_i$，存储其作差之后的$a_i$（其中，应有一$a_i$小于等于0，有一$a_i$等于0，和一$a_i$大于等于0），并把原第二大的$a_i$（即作差后等于0的$a_i$存入$ans$）。

然后对于各i的最大$a_i$做排序后按从大到小优先做贪心，累加进$ans$，并在此过程中记录会不会有部门人数超过$\frac{n}{2}$。如果直到程序结束各计数器都没有超过$\frac{n}{2}$，那么直接输出$ans$即可。否则，应当跳出贪心，在不考虑被占满部门的情况下重新对未使用的i的最大$a_i$求和，再累加进$ans$，即是最终答案。由于已经有一个部门被$\frac{n}{2}$占满了，剩下的人数一定小于等于$\frac{n}{2}$，所以可以直接对最大值求和。
### 代码
```cpp
#include<bits/stdc++.h>
using namespace std;
long long t;
struct P{
    long long a,b,c;
    long long maxabc;
}p[1000005];
bool cmp(P x,P y){
    return x.maxabc>y.maxabc;
}
int main (){
    cin>>t;
    while(t--){
        long long ans=0,n=0;
        cin>>n;
        for(int i=1;i<=n;i++){
            long long x=0,y=0,z=0,midxyz=0;
            cin>>x>>y>>z;
            if((x>=y&&y>=z)||(z>=y&&y>=x))midxyz=y;
            else if((x>=z&&z>=y)||(y>=z&&z>=x))midxyz=z;
            else if((y>=x&&x>=z)||(z>=x&&x>=y))midxyz=x;
            x-=midxyz;y-=midxyz;z-=midxyz;
            ans+=midxyz;
            p[i].a=x;p[i].b=y;p[i].c=z;p[i].maxabc=max({x,y,z});
        }
        sort(p+1,p+1+n,cmp);
        long long cnta=0,cntb=0,cntc=0,scnt=0;
        for(int i=1;i<=n;i++){
            if(cnta>=n/2||cntb>=n/2||cntc>=n/2)break;
            if(p[i].maxabc==p[i].a)cnta++;
            if(p[i].maxabc==p[i].b)cntb++;
            if(p[i].maxabc==p[i].c)cntc++;
            ans+=p[i].maxabc;
            scnt=i;
        }
        if(scnt==n){
            cout<<ans<<endl;
            continue;
        }else{
            for(int i=scnt;i<=n;i++){ //实际上这儿的for循环可以挪到上面
                if(cnta>=n/2){
                    ans+=max({p[i].b,p[i].c});
                }else if(cntb>=n/2){
                    ans+=max({p[i].a,p[i].c});
                }else if(cntc>=n/2){
                    ans+=max({p[i].b,p[i].a});
                }
            }
        }
        cout<<ans<<endl;
    }
}
```
